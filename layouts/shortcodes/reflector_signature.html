{{- /* Host-aware token link for a single Reflector signature token, looked up by name within a clone's `signatures` list in data/unispring.yml. This is the brand-aligned name. The `reflector_issue` shortcode is kept as an alias so pages written under the previous noun keep working; both shortcodes operate on the same data field and emit identical HTML. Args: clone - clone key under unispring.reflector.clones (e.g. "1xETH") name - signature-token name to match within that clone's signatures list show - link-text source when `text` is unset: "name" (default) or "address" text - explicit link text (overrides `show`) Emits a token explorer link (text wrapped in ) when the signature token has an address; otherwise "(pending deploy)". Rendered as raw HTML so it works inside prose and markdown table cells. */ -}} {{- $clone := index hugo.Data.unispring.reflector.clones (.Get "clone") -}} {{- $name := .Get "name" -}} {{- $sig := dict -}} {{- range $clone.signatures -}} {{- if eq .name $name -}}{{- $sig = . -}}{{- end -}} {{- end -}} {{- $addr := index $sig "address" -}} {{- if $addr -}} {{- $text := .Get "text" -}} {{- if not $text -}} {{- if eq (.Get "show" | default "name") "address" -}}{{- $text = $addr -}}{{- else -}}{{- $text = index $sig "name" -}}{{- end -}} {{- end -}} {{- $host := partial "explorer-host" "" -}} {{- $url := printf "https://%s/token/%s" $host $addr -}} {{ $text | safeHTML }} {{- else -}} (pending deploy) {{- end -}}
