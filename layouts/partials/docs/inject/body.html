{{- /* Client-side block-explorer host switcher. Every contract has the same address on every chain, so switching network only swaps the link host (e.g. arbiscan.io -> etherscan.io). This script: - reads the config JSON emitted below (from data/explorers.yml), - on load, sets every explorer link to the saved/default network, - on selector change, rewrites all explorer links and persists the choice. Links whose URL contains a `skip` address are never touched. Links flagged data-esname also get their visible text set to the network's explorer name. */ -}} {{- $cfg := hugo.Data.explorers -}} {{- $hosts := slice -}} {{- range $k, $n := $cfg.networks -}}{{- $hosts = $hosts | append $n.host -}}{{- end -}} {{- $live := slice -}} {{- range $cfg.order -}} {{- $n := index $cfg.networks . -}} {{- if $n.live -}}{{- $live = $live | append (dict "key" . "name" $n.name "host" $n.host) -}}{{- end -}} {{- end -}} {{- $config := dict "default" $cfg.default "hosts" $hosts "networks" $live "skip" (default (slice) $cfg.skip) -}}
