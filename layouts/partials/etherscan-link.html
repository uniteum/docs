{{- /* Build a block-explorer link. Args (dict): address, section, text, network (optional network key, e.g. "mainnet"), type ("address"|"token"|"tx"). The host comes from data/explorers.yml (default network), or from `network` if given. The address path is identical on every chain, so the network selector dropdown can rewrite the host client-side — see the `data-eslink` marker and assets/explorer-switch.js. */ -}} {{- $address := .address -}} {{- $section := .section -}} {{- $text := .text -}} {{- $network := .network -}} {{- $type := .type | default "address" -}} {{- $host := partial "explorer-host" $network -}} {{- $base := printf "https://%s" $host -}} {{- $url := "" -}} {{- if eq $type "tx" -}} {{- $url = printf "%s/tx/%s" $base $address -}} {{- else if eq $type "token" -}} {{- $url = printf "%s/token/%s" $base $address -}} {{- if $section -}}{{- $url = printf "%s#%s" $url $section -}}{{- end -}} {{- else -}} {{- $url = printf "%s/address/%s" $base $address -}} {{- if $section -}}{{- $url = printf "%s#%s" $url $section -}}{{- end -}} {{- end -}} {{- $linkText := $address -}} {{- if $text -}}{{- $linkText = $text -}}{{- end -}} {{ $linkText | safeHTML }}
